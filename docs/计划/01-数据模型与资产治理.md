### 1.3.1 Bug 扩展（在现模型上追加字段）

```prisma
model Bug {
  // ...既有字段全部保留
  assigneeId   String?   @map("assignee_id")     // 指派的用户 ID（可空=未指派）
  priority     String    @default("medium")      // low | medium | high | urgent
  dueDate      DateTime? @map("due_date")        // 截止日
  labels       String[] // 原生数组：["ui","p0"]（§1.3.0 已由 JSON 字符串升级）

  @@index([projectId, assigneeId])
  @@index([projectId, priority])
}
```

### 1.3.2 新增四个模型

```prisma
/// 缺陷评论 / 活动流：人与 AI 都能追加，形成对话闭环
model BugComment {
  id         String   @id
  bugId      String   @map("bug_id")
  authorType String   @map("author_type")   // user | ai
  authorId   String?  @map("author_id")    // user id 或 api_key id
  content    String
  createdAt  DateTime @default(now()) @map("created_at")

  @@index([bugId, createdAt])
  @@map("bug_comments")
}

/// 缺陷模板：新团队默认三个模板（缺陷/需求/线上事故），可自定义
model BugTemplate {
  id            String   @id
  name          String
  titleTemplate String   @map("title_template")     // 支持 {module} 占位
  fields        String                            // JSON：默认 severity/labels/steps 骨架
  createdBy     String?  @map("created_by")
  createdAt     DateTime @default(now()) @map("created_at")

  @@map("bug_templates")
}

/// 保存的视图：Excel 式「我的筛选」——筛选+排序+列显隐按用户持久化
model SavedView {
  id        String   @id
  userId    String   @map("user_id")
  entity    String                              // bug | file | task
  name      String
  filters   String                              // JSON
  sort      String                              // JSON
  columns   String                              // JSON：列显隐与顺序
  isDefault Boolean  @default(false) @map("is_default")
  createdAt DateTime @default(now()) @map("created_at")

  @@unique([userId, entity, name])
  @@map("saved_views")
}

/// 项目归档（软删除前置）：归档项目不显示在看板但数据可恢复
model Project {
  // ...既有字段保留
  archivedAt DateTime? @map("archived_at")
  createdBy  String?   @map("created_by")
}

/// 语义索引（pgvector）：bug/note 的 embedding，provider 默认 none 时本表闲置
/// 注意： Prisma 无 vector 类型，migration 中用 $executeRaw 追加向量列与 HNSW 索引：
///   ALTER TABLE "Embedding" ADD COLUMN "embedding" vector(1536);
///   CREATE INDEX ON "Embedding" USING hnsw ("embedding" vector_cosine_ops);
model Embedding {
  id         String   @id
  entityType String                 // bug | note
  entityId   String
  model      String                 // text-embedding-3-small / bge-m3 等
  dim        Int      @default(1536)
  createdAt  DateTime @default(now()) @map("created_at")
  updatedAt  DateTime @updatedAt @map("updated_at")

  @@unique([entityType, entityId, model])
  @@index([entityType])
  @@map("embeddings")
}
```

### 1.3.3 资产治理物理结构

```
data/
├── attachments/     # 在役附件：<yyyy-mm>/<attachmentId>.<ext>
├── derived/         # 图片降采样缓存
└── trash/           # 回收区：实体删除后的文件在此保留 ATTACHMENT_TRASH_DAYS 天
```

- `moveToTrash(storagePath)`：把文件移入 `trash/<原相对路径>`，保持可恢复。
- `purgeOlderThan(days)`：清理超期回收区文件（MCP 工具 `purge_trash` 或启动时自动执行一次）。

## 1.4 关键实现契约

### config.ts 增量

```ts
export const paths = {
  uploads: path.join(config.dataDir, 'attachments'),
  derived: path.join(config.dataDir, 'derived'),
  trash:   path.join(config.dataDir, 'trash'),
};
export const config = {
  // ...既有
  attachmentTrashDays: Number(process.env.ATTACHMENT_TRASH_DAYS ?? 30),
  jwtSecret: process.env.JWT_SECRET ?? '',
  registrationOpen: (process.env.REGISTRATION_OPEN ?? 'true') !== 'false',
  accessTokenTtlSec: 15 * 60,
  refreshTokenTtlDays: 30,
};
```

### storage.ts 增量签名

```ts
async moveToTrash(storagePath: string): Promise<string>   // 返回 trash 内新路径
async purgeOlderThan(days: number): Promise<number>      // 返回清理文件数
```
